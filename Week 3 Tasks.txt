public class SanjayMarsWeightCalculator {

    // variables declared at the top
    static float earthWeightBalakrishnan = 84.0f;
    static float marsGravityFactorBalakrishnan = 0.38f;
    static float marsWeightBalakrishnan;

    public static void main(String[] args) {

        // calculate Mars weight (float)
        marsWeightBalakrishnan = earthWeightBalakrishnan * marsGravityFactorBalakrishnan;
        System.out.println("Mars weight (float): " + marsWeightBalakrishnan);


        double marsWeightDoubleBalakrishnan = marsWeightBalakrishnan;
        System.out.printf("Mars weight as double (4 decimals): %.4f\n", marsWeightDoubleBalakrishnan);


        int marsWeightIntBalakrishnan = (int) marsWeightDoubleBalakrishnan;
        System.out.println("Mars weight converted to int: " + marsWeightIntBalakrishnan);


        char marsWeightCharBalakrishnan = (char) marsWeightIntBalakrishnan;
        System.out.println("Int converted to char: " + marsWeightCharBalakrishnan);

        int charMathBalakrishnan = marsWeightCharBalakrishnan + 10;
        System.out.println("Char + 10 converted to int: " + charMathBalakrishnan);

// The Numbers are my weight in Kilograms

    }
}

import java.util.Random;

public class EvenorOddSanjay {

    public static void main(String[] args) {

        Random rand = new Random();

        int numberBalakrishnan = rand.nextInt(100) + 1;

        System.out.println("Generated random number: " + numberBalakrishnan);

        if (numberBalakrishnan % 2 == 0) {
            System.out.println("The number is EVEN.");
        } else {
            System.out.println("The number is ODD.");
        }
    }
}

public class LegoBricks {

    public static void main(String[] args) {

        int amountOfBricks = 68; // odd between 50–100
        int containerCapacity = 6; // even between 5–10

        int fullContainers = amountOfBricks / containerCapacity;
        int remainingBricks = amountOfBricks % containerCapacity;

        int totalContainers = fullContainers;

        if (remainingBricks > 0) {
            totalContainers++;
        }

        System.out.println("Total bricks: " + amountOfBricks);
        System.out.println("Container capacity: " + containerCapacity);
        System.out.println("Full containers: " + fullContainers);
        System.out.println("Remaining bricks: " + remainingBricks);
        System.out.println("Total containers needed: " + totalContainers);
    }
}
